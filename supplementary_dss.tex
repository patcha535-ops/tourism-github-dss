\section{Overview}
This file reproduces in full the code that generates every table, figure and number of the manuscript. Python $\ge3.10$ with \texttt{numpy}, \texttt{scipy}, \texttt{pandas} and \texttt{matplotlib}; \texttt{pdftotext} is used once to read the compiled data tables. Run \texttt{sh run\_all\_dss.sh data\_tourism\_2024.pdf} (about one minute); the optional first step that rebuilds \texttt{panel.csv} from the three monthly Ministry workbooks is described in that script.

\paragraph{Data.} \texttt{provinces\_2024.csv} (77 rows) contains, for each province, recorded visits and receipts (total, Thai, foreign), accommodation guests and occupancy (Ministry of Tourism and Sports revised workbook, 28 August 2025); rooms, length of stay and daily expenditure (Domestic Tourism Statistics 2024); GPP (NESDC); registered population and area (National Statistical Office); and province centroids (OCHA boundaries). \texttt{build\_dataset.py} checks that Thai plus foreign visits and receipts equal the totals, that provincial values reconcile with the published regional and national rows (visits, receipts, guests, rooms), and reports the GPP rounding discrepancy of THB 4 million. SHA-256 of \texttt{provinces\_2024.csv}: \texttt{6bbd28cd9bb1210cb80c9d8af57b129cf0577c7ff27399ef16cab043d51f190a}.

\paragraph{Run.} Identifier \texttt{8671afc4b107}; seed 20260101; 10{,}000 bootstrap replications (resampling provinces within regions); 9{,}999 permutations for global and local Moran statistics; local classification stability over 20 independent permutation seeds.

\paragraph{Decision module.} \texttt{te\_decision.py} reads \texttt{provinces\_2024.csv} and \texttt{panel.csv} (SHA-256 of \texttt{panel.csv}: \texttt{4e54ae4da5df6d74a056a1b31e4a7a54c134119270fd52d6135b1961cc9e9029}); province names in the two sources are matched through the 2024 annual visit totals, which are unique across provinces. Run identifier \texttt{0ddfd68b0da1}. The module is deterministic (no random numbers).

# ----------------------------------------------------------------------------- measures (Eq. 1-9)
def punch_score(R, V):
    """Eq. 1: PS_i = 100 * (R_i / R) / (V_i / V) = 100 * y_i / ybar."""
    return 100.0 * (R / R.sum()) / (V / V.sum())


def gini(x):
    """Eq. 3, unweighted."""
    x = np.sort(np.asarray(x, float))
    n = len(x)
    return 2.0 * np.sum(np.arange(1, n + 1) * x) / (n * x.sum()) - (n + 1.0) / n


def wgini(x, w):
    """Population-weighted Gini of per-capita variable x with weights w (trapezoid rule on the weighted Lorenz curve)."""
    o = np.argsort(x, kind="stable")
    x, w = np.asarray(x, float)[o], np.asarray(w, float)[o]
    p = w / w.sum()
    inc = x * w / (x * w).sum()
    L = np.concatenate([[0.0], np.cumsum(inc)])
    P = np.concatenate([[0.0], np.cumsum(p)])
    return 1.0 - np.sum((L[1:] + L[:-1]) * (P[1:] - P[:-1]))


def theil(x):
    """Eq. 4."""
    x = np.asarray(x, float)
    s = x / x.sum()
    s = s[s > 0]
    return float(np.sum(s * np.log(len(x) * s)))


def theil_decomp(x, g):
    """Eq. 5-7: returns (T, T_between, T_within) with g an array of group labels."""
    x, g = np.asarray(x, float), np.asarray(g)
    n, X = len(x), x.sum()
    tb = tw = 0.0
    for lab in np.unique(g):
        m = g == lab
        S = x[m].sum() / X
        tb += S * np.log(S / (m.sum() / n))
        tw += S * theil(x[m])
    return tb + tw, tb, tw


def topk(x, k):
    return float(np.sort(x)[-k:].sum() / x.sum())


def shape_stats(x, g):
    T, tb, tw = theil_decomp(x, g)
    return gini(x), T, 100.0 * tb / T


# ----------------------------------------------------------------------------- spatial
def haversine(lat, lon):
    lat, lon = np.radians(lat), np.radians(lon)
    dlat, dlon = lat[:, None] - lat[None, :], lon[:, None] - lon[None, :]
    a = np.sin(dlat / 2) ** 2 + np.cos(lat[:, None]) * np.cos(lat[None, :]) * np.sin(dlon / 2) ** 2
    return 2 * 6371.0088 * np.arcsin(np.sqrt(a))


def rowstd(W):
    s = W.sum(1, keepdims=True)
    return np.divide(W, s, out=np.zeros_like(W), where=s > 0)


def weights(D, kind, band=None, k=5):
    n = len(D)
    D = D.copy()
    np.fill_diagonal(D, np.inf)
    if kind == "idw":
        W = 1.0 / D
    elif kind == "knn":
        W = np.zeros((n, n))
        idx = np.argsort(D, axis=1)[:, :k]
        np.put_along_axis(W, idx, 1.0, axis=1)
    else:
        W = (D <= band).astype(float)
    return rowstd(W)


def moran(x, W):
    z = x - x.mean()
    return float(z @ W @ z / (z @ z))                 # row-standardised: W0 = n


def moran_perm(x, W, B, rng):
    I = moran(x, W)
    z = x - x.mean()
    perm = np.array([moran(z[rng.permutation(len(z))], W) for _ in range(B)])
    zsc = (I - perm.mean()) / perm.std(ddof=1)
    p = (1 + np.sum(np.abs(perm - perm.mean()) >= abs(I - perm.mean()))) / (B + 1)
    return I, float(zsc), float(p)


def local_moran(x, W, B, rng, k=5):
    """Local Moran's I_i = z_i * (W z)_i with z standardised (sum z^2 / n = 1) and conditional permutation inference: the focal
    value is held fixed and the k neighbour values are drawn without replacement from the other n-1 values (row-standardised
    weights, so the spatial lag is a mean of k values).  Pseudo-p = (1 + #{|I_i^sim| >= |I_i|}) / (B + 1).
    Quadrants: 1 High-High, 2 Low-High, 3 Low-Low, 4 High-Low (own value vs spatial lag)."""
    n = len(x)
    z = (x - x.mean()) / x.std()
    lag = W @ z
    Ii = z * lag
    quad = np.where(z >= 0, np.where(lag >= 0, 1, 4), np.where(lag >= 0, 2, 3))
    p = np.zeros(n)
    for i in range(n):
        others = np.delete(z, i)
        idx = np.argpartition(rng.random((B, n - 1)), k, axis=1)[:, :k]
        sim = z[i] * others[idx].mean(1)
        p[i] = (1 + np.sum(np.abs(sim) >= abs(Ii[i]))) / (B + 1)
    return quad, Ii, p

def tex_table(name, outdir, spec, header, rows, long=False):

    else:
        s = f"\\begin{{tabular}}{{{spec}}}\n\\toprule\n{header} \\\\\n\\midrule\n{body}\n\\bottomrule\n\\end{{tabular}}\n"
    open(f"{outdir}/tex/{name}.tex", "w", encoding="utf-8").write(s)

def read_workbook(path):
    wb = openpyxl.load_workbook(path, read_only=True, data_only=True)
    out = []
    for name in [s for s in wb.sheetnames if SHEET.match(s)]:
        rows = []
        for r in wb[name].iter_rows(values_only=True):
            if len(r) > 1 and isinstance(r[1], str):
                nums = [x for x in r[2:] if isinstance(x, (int, float))]
                if len(nums) >= 24:
                    rows.append((r[1].strip(), nums[:24]))
        out.append(rows)
    return out                                                       # list over months of [(name, 24 numbers)]


def split(rows):
    prov = [(n, v) for n, v in rows if not n.startswith(REGION_PREFIX) and n != TOTAL_ROW]
    total = [v for n, v in rows if n == TOTAL_ROW][0]

import hashlib
import json
import os
import platform
import sys

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import scipy
from scipy import stats

LAMBDA_GRID = np.round(np.arange(0.0, 2.0 + 1e-9, 0.05), 10)      # risk-aversion grid used for the robustness classification
LAMBDA_SHOW = [0.0, 1.0, 2.0]                                       # columns of the priority table (lambda = 1 is the PSDF baseline)
ROBUST_SHARE, TOP_K = 0.90, 10


# ----------------------------------------------------------------------------- core definitions
def punch_score(R, V):
    return 100.0 * (R / R.sum()) / (V / V.sum())                    # Eq. (1)


def pct_rank(x):
    return (stats.rankdata(x) - 1.0) / (len(x) - 1.0)               # average ranks scaled to [0,1], as in the PSDF application


def adjusted_score(ps, r, lam):
    """Risk-adjusted Punch Score  S_i(lambda) = PS_i / (1 + lambda r_i)  (single-criterion instance of Z w / (1 + lambda R))."""
    return ps / (1.0 + lam * r)


def pareto_front(ps, r):
    """Indices not dominated by another province with PS >= and risk <= (at least one strict)."""
    n = len(ps)
    keep = []
    for i in range(n):
        dom = np.any((ps >= ps[i]) & (r <= r[i]) & ((ps > ps[i]) | (r < r[i])))
        if not dom:
            keep.append(i)
    return np.array(keep)


def crossing(ps_i, r_i, ps_j, r_j):
    """lambda at which S_i = S_j (None if no non-negative crossing): lambda* = (PS_j - PS_i) / (PS_i r_j - PS_j r_i)."""
    den = ps_i * r_j - ps_j * r_i
    if den == 0:
        return None
    lam = (ps_j - ps_i) / den
    return float(lam) if lam >= 0 else None


def leader_path(ps, r, lam_max=50.0, step=0.001):
    lam = np.arange(0.0, lam_max + step / 2, step)
    S = ps[None, :] / (1.0 + lam[:, None] * r[None, :])
    lead = S.argmax(1)
    path, start = [], 0
    for k in range(1, len(lam) + 1):
        if k == len(lam) or lead[k] != lead[start]:
            path.append((int(lead[start]), float(lam[start]), float(lam[k - 1]) if k < len(lam) else float(lam_max)))
            start = k
    return path


# ----------------------------------------------------------------------------- data
def load(prov_csv, panel_csv):
    df = pd.read_csv(prov_csv)
    pn = pd.read_csv(panel_csv)
    pn["year"] = pn.month.str[:4].astype(int)
    tot24 = pn[pn.year == 2024].groupby("province").visitors_total.sum()
    key = {int(round(v)): p for p, v in tot24.items()}
    assert len(key) == 77, "2024 annual visits must be unique across provinces to match the two sources"
    df["panel_name"] = [key[int(round(v))] for v in df.visits]
    assert df.panel_name.nunique() == 77
    def monthly(year):
        sub = pn[pn.year == year].pivot(index="province", columns="month", values="visitors_total")
        return sub.loc[df.panel_name].values
    def annual(year, col):
        return pn[pn.year == year].groupby("province")[col].sum().loc[df.panel_name].values
    return df, monthly(2024), monthly(2023), annual(2023, "visitors_total"), annual(2023, "receipts_total")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("prov"); ap.add_argument("panel")
    ap.add_argument("--outdir", default="results_dss")
    a = ap.parse_args()

    names = df.province.values
    R, V = df.receipts.values, df.visits.values

    # ---- Proposition 1: consistent aggregation of the Punch Score
    nat = float(np.sum(V / V.sum() * ps))
    reg_gap = 0.0
    for g in ["Bangkok", "Central", "East", "North", "Northeast", "South"]:
        m = (df.region == g).values
        direct = 100 * (R[m].sum() / R.sum()) / (V[m].sum() / V.sum())
        agg = float(np.sum(V[m] / V[m].sum() * ps[m]))
        reg_gap = max(reg_gap, abs(direct - agg))
    res["aggregation"] = dict(national_weighted_mean=nat, max_abs_region_gap=reg_gap)

    # ---- Proposition 2: lever identity and typology (Eq. 8 with 77-province national benchmarks)
    fT = df.visits_foreign.sum() / V.sum()

    yF = df.receipts_foreign.sum() * 1e6 / df.visits_foreign.sum()

    fi = df.visits_foreign.values / V

    ybar = R.sum() * 1e6 / V.sum()
    res["lever_identity_max_abs_error_thb"] = float(np.max(np.abs((y - ybar) - (comp + within))))
    yTi_ = df.receipts_thai.values * 1e6 / df.visits_thai.values
    yFi_ = df.receipts_foreign.values * 1e6 / df.visits_foreign.values
    slope = yFi_ - yTi_                                                   # d y_i / d f_i with provincial segment yields held fixed
    res["lever_slope"] = dict(n_positive=int((slope > 0).sum()), min_slope_thb=float(slope.min()), max_slope_thb=float(slope.max()),
                              min_ratio=float((yFi_ / yTi_).min()), max_ratio=float((yFi_ / yTi_).max()), median_ratio=float(np.median(yFi_ / yTi_)))

def test_p1_aggregation_random():
    rng = np.random.default_rng(1)
    for _ in range(200):
        n = rng.integers(3, 40)
        V, R = rng.uniform(1, 100, n), rng.uniform(1, 500, n)
        ps = D.punch_score(R, V)
        assert abs(np.sum(V / V.sum() * ps) - 100) < 1e-9                    # national mean is 100
        g = rng.random(n) < 0.5
        if g.any() and (~g).any():
            direct = 100 * (R[g].sum() / R.sum()) / (V[g].sum() / V.sum())
            assert abs(direct - np.sum(V[g] / V[g].sum() * ps[g])) < 1e-9     # group score = visit-weighted mean of member scores


def test_p2_lever_identity_random():
    rng = np.random.default_rng(2)
    for _ in range(200):
        n = 20
        VT, VF = rng.uniform(1, 100, n), rng.uniform(1, 100, n)
        RT, RF = VT * rng.uniform(1, 5, n), VF * rng.uniform(5, 30, n)
        V, R = VT + VF, RT + RF
        f = VF.sum() / V.sum(); yT, yF = RT.sum() / VT.sum(), RF.sum() / VF.sum()
        fi = VF / V
        comp = (fi - f) * (yF - yT)
        within = (1 - fi) * (RT / VT - yT) + fi * (RF / VF - yF)
        assert np.allclose(R / V - R.sum() / V.sum(), comp + within, atol=1e-9)


def test_p3_p4_comparative_statics_random():
    rng = np.random.default_rng(3)
    for _ in range(500):
        n = rng.integers(5, 60)
        ps, r = rng.uniform(10, 500, n), rng.uniform(0, 1, n)
        front = set(D.pareto_front(ps, r))
        lams = np.sort(rng.uniform(0, 5, 8))
        lead = [int(np.argmax(D.adjusted_score(ps, r, l))) for l in lams]
        assert all(i in front for i in lead)                                  # P3: maximisers are Pareto-efficient
        assert all(r[a] >= r[b] - 1e-12 for a, b in zip(lead[:-1], lead[1:]))  # P4: risk of the maximiser never rises with lambda


def test_crossing_formula():
    rng = np.random.default_rng(4)
    for _ in range(500):
        ps_i, ps_j = rng.uniform(50, 500, 2); r_i, r_j = rng.uniform(0, 1, 2)
        lam = D.crossing(ps_i, r_i, ps_j, r_j)
        if lam is not None:
            assert abs(D.adjusted_score(ps_i, r_i, lam) - D.adjusted_score(ps_j, r_j, lam)) < 1e-8
